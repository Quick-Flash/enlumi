pub fn filter(
    time: &[f64],
    inputs: &[&[f64]],
    outputs: &mut [&mut [f64]],
    _state: &mut State,
) {
    // Inputs follow the Inputs list order. Samples are aligned by timestamp.
    // time[i] is log time in seconds; NaN represents a gap.
    const GAIN: f64 = 1.0;
    const OFFSET: f64 = 0.0;

    for i in 0..time.len() {
        // Default: sum every input. One input passes through unchanged.
        let value: f64 = inputs.iter().map(|field| field[i]).sum();

        // let mut value = 0.0;
        // for input_index in 0..inputs.len() {
        //     value += inputs[input_index][i];
        // }
        outputs[0][i] = value * GAIN + OFFSET;

        // let value = inputs[0][i] / inputs[1][i]; // signal / parameter

    }
